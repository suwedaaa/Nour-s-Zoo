cmake_minimum_required(VERSION 3.15)
project(NoursZooDesktop LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The desktop adapter currently supports Windows.")
endif()
add_executable(NoursZoo desktop/Simulator.cpp src/main.cpp src/Game.cpp src/Renderer.cpp src/Input.cpp src/Sound.cpp src/ButtonState.cpp src/Dialogue.cpp src/DialogueData.cpp src/DialogueHistory.cpp src/Conversation.cpp src/StudyTimer.cpp src/Save.cpp src/SaveData.cpp src/Power.cpp src/PowerPolicy.cpp)
target_compile_features(NoursZoo PRIVATE cxx_std_17)
target_compile_definitions(NoursZoo PRIVATE NOURS_DESKTOP NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_include_directories(NoursZoo PRIVATE desktop/compat desktop include)
target_link_libraries(NoursZoo PRIVATE user32 gdi32 winmm)
enable_testing()
add_test(NAME desktop_smoke COMMAND NoursZoo --self-test)
set_tests_properties(desktop_smoke PROPERTIES ENVIRONMENT "NOURS_SAVE_PATH=${CMAKE_BINARY_DIR}/smoke-save.bin")
add_test(NAME system_flow COMMAND NoursZoo --system-test)
set_tests_properties(system_flow PROPERTIES ENVIRONMENT "NOURS_SAVE_PATH=${CMAKE_BINARY_DIR}/system-test-save.bin")
add_test(NAME boot_wipe COMMAND NoursZoo --boot-wipe-test)
set_tests_properties(boot_wipe PROPERTIES ENVIRONMENT "NOURS_SAVE_PATH=${CMAKE_BINARY_DIR}/boot-test-save.bin")
add_executable(button_tests test/input/button_state_test.cpp src/ButtonState.cpp)
add_executable(dialogue_tests test/dialogue/dialogue_test.cpp src/Dialogue.cpp src/DialogueData.cpp src/DialogueHistory.cpp src/Conversation.cpp)
add_executable(study_tests test/study/study_test.cpp src/StudyTimer.cpp)
add_executable(sound_tests test/sound/sound_test.cpp src/Sound.cpp)
add_executable(save_tests test/save/save_test.cpp src/Save.cpp src/SaveData.cpp)
target_compile_definitions(save_tests PRIVATE NOURS_DESKTOP NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
add_executable(power_tests test/power/power_test.cpp src/PowerPolicy.cpp src/StudyTimer.cpp)
target_include_directories(sound_tests BEFORE PRIVATE test/sound)
foreach(test_target button_tests dialogue_tests study_tests sound_tests save_tests power_tests)
  target_include_directories(${test_target} PRIVATE include)
  target_compile_features(${test_target} PRIVATE cxx_std_17)
  target_compile_options(${test_target} PRIVATE /UNDEBUG)
  add_test(NAME ${test_target} COMMAND ${test_target})
endforeach()
